estates_beduins_sand_routes = {
    icon = estate_burghers_arb_oasis_development
    max_absolutism = -2.5
    conditional_modifier = { trigger = { has_absolutism_reduction_for_estate_privileges = yes } modifier = { max_absolutism = 1 } }
    influence = 0.1
	loyalty = 0.05
    check_valid_when_tag_switching = yes
    is_valid = {
        any_owned_province = {
            OR = {
                has_terrain = desert
				has_terrain = coastal_desert
                has_climate = arid
            }
        }
    }

    can_select = {
        any_owned_province = {
            OR = {
                has_terrain = desert
				has_terrain = coastal_desert
                has_climate = arid
            }
            OR = { 
                  region = maghreb_region
                  region = arabia_region
                  region = sahel_region  
                }

        }
    }

    penalties = {
	}

    on_granted = {
		
		custom_tooltip = beduin_sand_routes_tooltip_gt
		hidden_effect = {
			every_owned_province = {
				limit = {
					OR = {
						has_terrain = desert
						has_terrain = coastal_desert
                        has_climate = arid
					}
                    OR = { 
                        region = maghreb_region
                        region = arabia_region
                        region = sahel_region  
                    }
				}
				add_province_modifier = {
					name = beduin_sand_routes_modifier
					duration = -1
				}
			}
		}
	}
	on_revoked = {
		
		custom_tooltip = beduin_sand_routes_tooltip_rt
		hidden_effect = {
			every_owned_province = {
				limit = {
					has_province_modifier = beduin_sand_routes_modifier
				}
				remove_province_modifier = beduin_sand_routes_modifier
			}
		}
	}

    ai_will_do = {
		factor = 1
	}
}
