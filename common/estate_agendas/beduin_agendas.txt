estate_beduin_hire_advisor = {
	can_select = {
		NOT = {
			employed_advisor = {
				category = MIL
			}
		}
	}
	selection_weight = {
		factor = 1
	}
	task_requirements = {
		if = {
			limit = {
				has_estate_agenda_auto_completion = {
					estate = beduini_estate
				}
			}
			has_estate_agenda_auto_completion = {
				estate = beduini_estate
			}
		}
		else = {
			employed_advisor = {
				category = MIL
			}
			is_in_deficit = no
		}
	}
    task_completed_effect = {
		on_completed_agenda_effect = {
			estate = beduini_estate
		}
		add_estate_loyalty = {
			estate = beduini_estate
			loyalty = 10
		}
    }
    failing_effect = {
		on_failed_agenda_effect = yes
		clr_auto_complete_flag = yes
        add_estate_loyalty_modifier = {
            estate = beduini_estate
			desc = EST_VAL_AGENDA_DENIED
			loyalty = -5
			duration = 7300
        }
    }
}

estate_beduini_two_privileges = {
	can_select = {
		NOT = {
			num_of_estate_privileges = {
				estate = beduini_estate
				value = 2
			}
		}
	}
	selection_weight = {
		factor = 1
	}
	pre_effect = {
		set_variable = {
			which = estate_beduini_two_privileges_var
			value = 0
		}
		if = {
			limit = {
				num_of_estate_privileges = {
					estate = beduini_estate
					value = 1
				}
			}
			set_variable = {
				which = estate_beduini_two_privileges_var
				value = 1
			}
		}
		else = {
			set_variable = {
				which = estate_beduni_two_privileges_var
				value = 0
			}
		}
	}
	task_requirements = {
		if = {
			limit = {
				has_estate_agenda_auto_completion = {
					estate = beduini_estate
				}
			}
			has_estate_agenda_auto_completion = {
				estate = beduini_estate
			}
		}
		else = {
			if = {
				limit = {
					check_variable = {
						which = estate_beduini_two_privileges_var
						value = 1
					}
				}
				num_of_estate_privileges = {
					estate = beduini_estate
					value = 2
				}
			}
			else = {
				num_of_estate_privileges = {
					estate = beduini_estate
					value = 1
				}
			}
		}
	}
    task_completed_effect = {
		on_completed_agenda_effect = {
			estate = beduini_estate
		}
		add_estate_loyalty = {
			estate = beduini_estate
			loyalty = 20
		}
    }
    failing_effect = {
		on_failed_agenda_effect = yes
		clr_auto_complete_flag = yes
        add_estate_loyalty_modifier = {
            estate = beduini_estate
			desc = EST_VAL_AGENDA_DENIED
			loyalty = -5
			duration = 7300
        }
    }
}
